% PFG System Architecture - TikZ figure

\begin{figure}[t]
\centering
\begin{tikzpicture}[
    node distance=1.5cm,
    box/.style={rectangle, draw=black!60, fill=blue!5, thick, minimum width=2.8cm, minimum height=0.8cm, align=center, font=\small},
    arrow/.style={->, thick, >=stealth},
    label/.style={font=\tiny, midway, fill=white, inner sep=1pt}
]

% Source code
\node[box, fill=green!10] (source) {Django\\Source Code};

% PFG Extractor
\node[box, below=of source] (extractor) {PFG\\Extractor};

% PFG
\node[box, below=of extractor, fill=yellow!10] (pfg) {Program Flow\\Graph};

% GNN
\node[box, below left=1.5cm and 0.5cm of pfg, fill=purple!10] (gnn) {GNN\\Model};

% Validator
\node[box, below right=1.5cm and 0.5cm of pfg, fill=orange!10] (validator) {Structural\\Validator};

% Output
\node[box, below=3.5cm of pfg, fill=red!10] (output) {Vulnerability\\Findings};

% Arrows
\draw[arrow] (source) -- (extractor) node[label] {parse};
\draw[arrow] (extractor) -- (pfg) node[label] {build};
\draw[arrow] (pfg) -- (gnn) node[label, left] {featurize};
\draw[arrow] (pfg) -- (validator) node[label, right] {structure};
\draw[arrow] (gnn) -- (output) node[label, left] {predictions};
\draw[arrow] (validator) -- (output) node[label, right] {confirm};

\end{tikzpicture}
\caption{PFG system architecture. Django source code is parsed into a Program Flow Graph, which feeds both the GNN scorer and the structural validator. Only predictions confirmed by the validator are reported.}
\label{fig:architecture}
\end{figure}
